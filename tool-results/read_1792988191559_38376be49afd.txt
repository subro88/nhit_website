    45→function defaultSection(type: string): any {
    46→  switch (type) {
    47→    case 'text':
    48→      return { type, tag: '', heading: '', paragraphs: [''] };
    49→    case 'quote':
    50→      return { type, tag: '', heading: '', paragraphs: [''], signedBy: '', signedOrg: '' };
    51→    case 'officerCards':
    52→      return { type, cards: [{ label: '', labelColor: 'red', title: '', lines: [''] }] };
    53→    case 'steps':
    54→      return { type, tag: '', heading: '', items: [''] };
    55→    case 'table':
    56→      return { type, tag: '', heading: '', columns: ['Column 1', 'Column 2'], rows: [{ main: '', sub: '', cells: [''] }] };
    57→    case 'objectives':
    58→      return { type, tag: '', heading: '', items: [{ title: '', desc: '' }] };
    59→    case 'grid':
    60→      return { type, tag: '', heading: '', items: [{ title: '', type: '', desc: '' }] };
    61→    case 'approvals':
    62→      return { type, tag: '', heading: '', items: [{ title: '', status: 'Approved', id: '', body: '', details: '' }] };
    63→    case 'helplines':
    64→      return { type, boxes: [{ title: '', value: '', note: '', color: 'red' }] };
    65→    case 'banner':
    66→      return { type, heading: '', paragraphs: [''], note: '' };
    67→    default:
    68→      return { type };
    69→  }

   111→/* Small repeatable list of objects */
   112→function MiniList({
   113→  label,
   114→  items,
   115→  onChange,
   116→  def,
   117→  titleOf,
   118→  renderFields,
   119→}: {
   120→  label: string;
   121→  items: any[];
   122→  onChange: (v: any[]) => void;
   123→  def: any;
   124→  titleOf: (item: any, i: number) => string;
   125→  renderFields: (item: any, set: (patch: any) => void) => React.ReactNode;
   126→}) {
   127→  const [open, setOpen] = useState<number | null>(0);
   128→  return (
   129→    <div className="space-y-2">
   130→      <div className="flex items-center justify-between">
   131→        <span className="text-xs font-bold uppercase tracking-wider text-slate-600">
   132→          {label} ({items.length})
   133→        </span>
   134→        <button
   135→          type="button"
   136→          onClick={() => {
   137→            onChange([...items, JSON.parse(JSON.stringify(def))]);
   138→            setOpen(items.length);
   139→          }}
   140→          className="inline-flex items-center gap-1 text-xs font-bold text-[#9E0B0F] hover:underline"
   141→        >
   142→          <Plus className="w-3.5 h-3.5" /> Add
   143→        </button>
   144→      </div>
   145→      {items.map((item, i) => {
   146→        const isOpen = open === i;
   147→        return (
   148→          <div key={i} className="rounded-xl border border-slate-200 bg-slate-50/60 overflow-hidden">
   149→            <div className="flex items-center gap-1 px-3 py-2">
   150→              <button type="button" className="flex-1 min-w-0 text-left text-sm font-bold text-slate-700 truncate" onClick={() => setOpen(isOpen ? null : i)}>
   151→                {titleOf(item, i) || `Item ${i + 1}`}
   152→              </button>
   153→              <button type="button" title="Move up" disabled={i === 0} onClick={() => { const n = [...items]; [n[i], n[i - 1]] = [n[i - 1], n[i]]; onChange(n); }} className="p-1.5 text-slate-400 hover:text-slate-800 disabled:opacity-30">
   154→                <ChevronUp className="w-4 h-4" />
   155→              </button>
   156→              <button type="button" title="Move down" disabled={i === items.length - 1} onClick={() => { const n = [...items]; [n[i], n[i + 1]] = [n[i + 1], n[i]]; onChange(n); }} className="p-1.5 text-slate-400 hover:text-slate-800 disabled:opacity-30">
   157→                <ChevronDown className="w-4 h-4" />
   158→              </button>
   159→              <button type="button" title="Remove" onClick={() => { if (confirm('Remove this item?')) onChange(items.filter((_, j) => j !== i)); }} className="p-1.5 text-slate-300 hover:text-red-600">
   160→                <Trash2 className="w-4 h-4" />
   161→              </button>
   162→            </div>
   163→            {isOpen && (
   164→              <div className="px-4 pb-4 pt-1 space-y-3 border-t border-slate-100 bg-white">
   165→                {renderFields(item, (patch) => {
   166→                  const next = [...items];
   167→                  next[i] = { ...next[i], ...patch };
   168→                  onChange(next);
   169→                })}
   170→              </div>
   171→            )}
   172→          </div>
   173→        );
   174→      })}
   175→    </div>
   176→  );
   177→}

   390→export default function PagesEditor() {
   391→  const [data, setData] = useState<Record<string, any> | null>(null);
   392→  const [activeKey, setActiveKey] = useState(PAGE_LIST[0].key);
   393→  const [tab, setTab] = useState<'form' | 'json'>('form');
   394→  const [jsonText, setJsonText] = useState('');
   395→  const [loading, setLoading] = useState(true);
   396→  const [busy, setBusy] = useState(false);
   397→  const [msg, setMsg] = useState<{ ok: boolean; text: string } | null>(null);
   398→
   399→  useEffect(() => {
   400→    (async () => {
   401→      const res = await fetch('/api/admin/pages');
   402→      const json = await res.json();
   403→      if (res.ok) {
   404→        setData(json.pages);
   405→        setJsonText(JSON.stringify(json.pages, null, 2));
   406→      } else {
   407→        setMsg({ ok: false, text: json.error || 'Could not load pages' });
   408→      }
   409→      setLoading(false);
   410→    })();
   411→  }, []);
   412→
   413→  const page = data?.[activeKey] ?? {};
   414→
   415→  function setPage(next: any) {
   416→    setData((prev) => (prev ? { ...prev, [activeKey]: next } : prev));
   417→  }
   418→
   419→  function setHero(key: string, value: string) {
   420→    setPage({ ...page, [key]: value });
   421→  }
   422→
   423→  function updateSection(idx: number, next: any) {
   424→    const sections = [...(page.sections ?? [])];
   425→    sections[idx] = next;
   426→    setPage({ ...page, sections });
   427→  }
   428→
   429→  function moveSection(idx: number, dir: -1 | 1) {
   430→    const sections = [...(page.sections ?? [])];
   431→    const j = idx + dir;
   432→    if (j < 0 || j >= sections.length) return;
   433→    [sections[idx], sections[j]] = [sections[j], sections[idx]];
   434→    setPage({ ...page, sections });
   435→  }
   436→
   437→  async function save() {
   438→    setMsg(null);
   439→    let payload = data;
   440→
   441→    if (tab === 'json') {
   442→      try {
   443→        payload = JSON.parse(jsonText);
   444→      } catch (e: any) {
   445→        setMsg({ ok: false, text: `JSON error: ${e.message}` });
   446→        return;
   447→      }
   448→    }
   449→
   450→    setBusy(true);
   451→    const res = await fetch('/api/admin/pages', {
   452→      method: 'PUT',
   453→      headers: { 'Content-Type': 'application/json' },
   454→      body: JSON.stringify({ pages: payload }),
   455→    });
   456→    const json = await res.json();
   457→    setBusy(false);
   458→
   459→    if (res.ok) {
   460→      setData(payload as Record<string, any>);
   461→      if (tab === 'json') setJsonText(JSON.stringify(payload, null, 2));
   462→      setMsg({ ok: true, text: 'Saved! The live pages are updated.' });
   463→    } else {
   464→      setMsg({ ok: false, text: json.error || 'Could not save' });
   465→    }
   466→  }
   467→
   468→  if (loading) {
   469→    return (
   470→      <div className="bg-white rounded-2xl border border-slate-200 p-10 text-center text-sm text-slate-500">
   471→        Loading pages…
   472→      </div>
   473→    );
   474→  }
   475→
   476→  if (!data) {
   477→    return (
   478→      <div className="bg-white rounded-2xl border border-red-200 p-10 text-center text-sm text-red-700">
   479→        {msg?.text || 'Could not load pages.'}
   480→      </div>
   481→    );
   482→  }
   483→
   484→  const activePage = PAGE_LIST.find((p) => p.key === activeKey)!;
   485→  const sections = (page.sections ?? []) as any[];
   486→
   487→  return (
   488→    <div className="space-y-5">
   489→      {/* Page switcher */}
   490→      <div className="flex flex-wrap gap-2">
   491→        {PAGE_LIST.map((p) => (
   492→          <button
   493→            key={p.key}
   494→            onClick={() => setActiveKey(p.key)}
   495→            className={`px-3.5 py-2 rounded-xl text-xs font-bold transition border ${
   496→              p.key === activeKey
   497→                ? 'bg-[#9E0B0F] text-white border-[#9E0B0F] shadow'
   498→                : 'bg-white text-slate-600 border-slate-200 hover:border-slate-400'
   499→            }`}
   500→          >
   501→            {p.label}
   502→          </button>
   503→        ))}
   504→      </div>
   505→
   506→      {/* Toolbar */}
   507→      <div className="flex flex-wrap items-center justify-between gap-3">
   508→        <div className="inline-flex bg-white rounded-xl border border-slate-200 p-1 shadow-sm">
   509→          <button
   510→            onClick={() => {
   511→              setTab('form');
   512→              setJsonText(JSON.stringify(data, null, 2));
   513→            }}
   514→            className={`inline-flex items-center gap-1.5 px-4 py-2 rounded-lg text-xs font-bold transition ${
   515→              tab === 'form' ? 'bg-slate-900 text-white' : 'text-slate-500 hover:text-slate-900'
   516→            }`}
   517→          >
   518→            <FormInput className="w-3.5 h-3.5" /> Content Form
   519→          </button>
   520→          <button
   521→            onClick={() => setTab('json')}
   522→            className={`inline-flex items-center gap-1.5 px-4 py-2 rounded-lg text-xs font-bold transition ${
   523→              tab === 'json' ? 'bg-slate-900 text-white' : 'text-slate-500 hover:text-slate-900'
   524→            }`}
   525→          >
   526→            <FileJson className="w-3.5 h-3.5" /> Advanced JSON
   527→          </button>
   528→        </div>
   529→
   530→        <div className="flex items-center gap-2">
   531→          <a
   532→            href={activePage.href}
   533→            target="_blank"
   534→            rel="noopener noreferrer"
   535→            className="inline-flex items-center gap-1.5 px-4 py-2 rounded-xl text-xs font-bold text-slate-600 hover:text-slate-900 border border-slate-200 bg-white"
   536→          >
   537→            <ExternalLink className="w-3.5 h-3.5" /> View Live Page
   538→          </a>
   539→          <button
   540→            onClick={save}
   541→            disabled={busy}
   542→            className="inline-flex items-center gap-2 px-5 py-2 rounded-xl bg-[#9E0B0F] hover:bg-[#7B080B] text-white text-xs font-bold shadow disabled:opacity-60"
   543→          >
   544→            <Save className="w-3.5 h-3.5" />
   545→            {busy ? 'Saving…' : 'Save Changes'}
   546→          </button>
   547→        </div>
   548→      </div>
   549→
   550→      {msg && (
   551→        <div
   552→          className={`flex items-center gap-2 text-sm rounded-xl px-4 py-3 ${
   553→            msg.ok
   554→              ? 'text-emerald-700 bg-emerald-50 border border-emerald-200'
   555→              : 'text-red-700 bg-red-50 border border-red-200'
   556→          }`}
   557→        >
   558→          {msg.ok ? <CheckCircle2 className="w-4 h-4 shrink-0" /> : <AlertCircle className="w-4 h-4 shrink-0" />}
   559→          {msg.text}
   560→        </div>
   561→      )}
   562→
   563→      {tab === 'form' ? (
   564→        <div className="space-y-5">
   565→          {/* Banner */}
   566→          <section className="bg-white rounded-2xl border border-slate-200 shadow-sm p-5 sm:p-6 space-y-4">
   567→            <h3 className="text-sm font-extrabold text-slate-900 uppercase tracking-wider">Page Banner</h3>
   568→            <div className="grid sm:grid-cols-2 gap-4">
   569→              <Field label="Small Line Above Title (kicker)">
   570→                <input className={inputCls} value={page.kicker ?? ''} onChange={(e) => setHero('kicker', e.target.value)} />
   571→              </Field>
   572→              <Field label="Badge">
   573→                <input className={inputCls} value={page.badge ?? ''} onChange={(e) => setHero('badge', e.target.value)} />
   574→              </Field>
   575→            </div>
   576→            <Field label="Page Title">
   577→              <input className={inputCls} value={page.title ?? ''} onChange={(e) => setHero('title', e.target.value)} />
   578→            </Field>
   579→            <Field label="Banner Subtitle">
   580→              <textarea rows={2} className={inputCls} value={page.subtitle ?? ''} onChange={(e) => setHero('subtitle', e.target.value)} />
   581→            </Field>
   582→          </section>
   583→
   584→          {/* Sections */}
   585→          <div className="space-y-3">
   586→            <div className="flex items-center justify-between">
   587→              <h3 className="text-sm font-extrabold text-slate-900 uppercase tracking-wider">
   588→                Page Sections ({sections.length})
   589→              </h3>
   590→              <select
   591→                className="px-3 py-2 rounded-xl text-xs font-bold border border-slate-200 bg-white"
   592→                value=""
   593→                onChange={(e) => {
   594→                  if (!e.target.value) return;
   595→                  setPage({ ...page, sections: [...sections, defaultSection(e.target.value)] });
   596→                  e.target.value = '';
   597→                }}
   598→              >
   599→                <option value="">+ Add Section…</option>
   600→                {SECTION_TYPES.map((t) => (
   601→                  <option key={t.value} value={t.value}>
   602→                    {t.label}
   603→                  </option>
   604→                ))}
   605→              </select>
   606→            </div>
   607→
   608→            {sections.map((s, i) => (
   609→              <div key={i} className="bg-white rounded-2xl border border-slate-200 shadow-sm overflow-hidden">
   610→                <div className="flex items-center gap-1 px-4 py-2.5 bg-slate-50 border-b border-slate-100">
   611→                  <span className="flex-1 min-w-0 text-xs font-bold text-slate-700">
   612→                    {i + 1}. {SECTION_TYPES.find((t) => t.value === s.type)?.label ?? s.type}
   613→                    {s.heading ? <span className="text-slate-400 font-semibold"> — {s.heading}</span> : null}
   614→                  </span>
   615→                  <button type="button" title="Move up" disabled={i === 0} onClick={() => moveSection(i, -1)} className="p-1.5 text-slate-400 hover:text-slate-800 disabled:opacity-30">
   616→                    <ChevronUp className="w-4 h-4" />
   617→                  </button>
   618→                  <button type="button" title="Move down" disabled={i === sections.length - 1} onClick={() => moveSection(i, 1)} className="p-1.5 text-slate-400 hover:text-slate-800 disabled:opacity-30">
   619→                    <ChevronDown className="w-4 h-4" />
   620→                  </button>
   621→                  <button
   622→                    type="button"
   623→                    title="Remove section"
   624→                    onClick={() => {
   625→                      if (confirm(`Remove this ${s.type} section from the page?`)) {
   626→                        setPage({ ...page, sections: sections.filter((_, j) => j !== i) });
   627→                      }
   628→                    }}
   629→                    className="p-1.5 text-slate-300 hover:text-red-600"
   630→                  >
   631→                    <Trash2 className="w-4 h-4" />
   632→                  </button>
   633→                </div>
   634→                <div className="p-5 space-y-4">
   635→                  <SectionForm section={s} onChange={(next) => updateSection(i, next)} />
   636→                </div>
   637→              </div>
   638→            ))}
   639→          </div>
   640→        </div>
   641→      ) : (
   642→        <div className="bg-white rounded-2xl border border-slate-200 shadow-sm overflow-hidden">
   643→          <div className="flex items-center justify-between px-5 py-3 border-b border-slate-100 bg-slate-50">
   644→            <span className="text-xs font-bold text-slate-600 uppercase tracking-wider">Full pages data (JSON)</span>
   645→            <button
   646→              onClick={() => setJsonText(JSON.stringify(data, null, 2))}
   647→              className="inline-flex items-center gap-1.5 text-xs font-bold text-slate-500 hover:text-slate-900"
   648→            >
   649→              <RotateCcw className="w-3.5 h-3.5" /> Reset
   650→            </button>
   651→          </div>
   652→          <textarea
   653→            value={jsonText}
   654→            onChange={(e) => setJsonText(e.target.value)}
   655→            spellCheck={false}
   656→            className="w-full h-[560px] p-5 font-mono text-xs leading-relaxed text-slate-800 focus:outline-none resize-y bg-white"
   657→          />
   658→        </div>
   659→      )}
   660→    </div>
   661→  );
   662→}
